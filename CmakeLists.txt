cmake_minimum_required(VERSION 4.00)
project(ARenderer)

set(CMAKE_CXX_STANDARD 23)

# Vulkan
find_package(Vulkan REQUIRED)

# CMake module
include(FetchContent)
include(assets/CopyAssets.cmake)
include(assets/shaders/CompileShaders.cmake)

# GLFW
FetchContent_Declare(
  glfw
  GIT_REPOSITORY https://github.com/glfw/glfw.git
  GIT_TAG 3.5.1      # pick a stable version
)
set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(GLFW_INSTALL OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(glfw)

# GLM
FetchContent_Declare(
  glm
  GIT_REPOSITORY https://github.com/g-truc/glm.git
  GIT_TAG 1.0.3  # or newer release
)
set(GLM_TEST_ENABLE OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(glm)

# Main executable
add_executable(${PROJECT_NAME})
add_dependencies(${PROJECT_NAME} CopyAssets)
add_dependencies(${PROJECT_NAME} CompileShaders)

target_include_directories(${PROJECT_NAME} 
    PRIVATE
    includes
)

target_sources(${PROJECT_NAME}
    PRIVATE
    "source/main.cpp"
    "source/ARenderer.cpp"
    "source/Vertex.cpp"
    "source/SwapCHain.cpp"
    "source/Model.cpp"
    "source/context/Instance.cpp"
    "source/context/Device.cpp"
    "source/context/PhysicalDevice.cpp"
    "source/pipeline/RenderPass.cpp"
)

target_link_libraries(${PROJECT_NAME}
    Vulkan::Vulkan
    glfw
    glm::glm
)
